\begin{theorem87}
Let $f$ be a continuous function. Then
\[
   \frac{d}{dx} \int_a^x f(t)\, dt = f(x).
\]
\end{theorem87}
\begin{proof}
Subtraction and division undoes addition and multiplication.
\end{proof}

\begin{theorem87}
Let $f$ be a continuously differentiable function. Then
\[
   f(x) =  f(a) + \int_a^x f^\prime(t)\, dt .
\]
\end{theorem87}
\begin{proof}
With the right starting point, addition and multiplication undoes subtraction and division.
\end{proof}


\end{exploration}
